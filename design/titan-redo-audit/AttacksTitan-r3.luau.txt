local function snd(id, pos, gain, pitch, from, cut)
	local at = Instance.new("Attachment"); at.WorldPosition = pos; at.Parent = workspace.Terrain
	local s = Instance.new("Sound")
	s.SoundId = "rbxassetid://" .. id; s.Volume = gain * 2; s.PlaybackSpeed = pitch * (0.96 + math.random() * 0.08)
	s.RollOffMinDistance, s.RollOffMaxDistance = 10, 85
	local eq = Instance.new("EqualizerSoundEffect"); eq.LowGain, eq.MidGain, eq.HighGain = -6, 2, 0; eq.Parent = s
	s.Parent = at
	if from and from > 0 then s.TimePosition = from end
	s:Play()
	if cut then                                          -- (long clips: faded out over 0.3 s from `cut`, never a tail past the move)
		task.delay(cut, function()
			local v0 = s.Volume
			for i = 1, 6 do task.wait(0.05); if not s.Parent then return end; s.Volume = v0 * (1 - i / 6) end
			s:Stop()
		end)
	end
	Debris:AddItem(at, (cut and cut + 0.6) or 8)
end

-- ===== the K meshes: clone, then place by the mesh's AUTHORED origin (attribute C = its centre in authored space; the
-- 3D importer turned every mesh 180 deg about Y). frame's -Z = the authored -Z. k = scale (number or Vector3), a = opacity
local function mesh(ctx, name)
	local t = GM:FindFirstChild(name)
	if not t then return nil end
	local p = t:Clone(); p.Transparency = 1; p.Parent = ctx.holder
	return { p = p, s0 = t.Size, c = t:GetAttribute("C") or Vector3.zero }
end
-- (measured 29 Oct, 4 Titans fighting: writing Size / CFrame / Transparency on every effect mesh every frame (cracks,
-- puddles and the flood hold still for seconds) cost ~half the frame rate, the same per-frame property churn as the old glow.
-- A write happens only when the value really changed; opacity in 0.05 steps)
local function put(m, frame: CFrame, k, a)
	if not m then return end
	local s = (typeof(k) == "Vector3") and k or Vector3.one * (k or 1)
	local tr = math.floor((1 - clamp01(a or 1)) * 20 + 0.5) / 20
	if tr ~= m.tr then m.tr = tr; m.p.Transparency = tr end
	if tr >= 1 then return end
	if not m.s or (m.s - s).Magnitude > 0.004 then m.s = s; m.p.Size = m.s0 * s end
	local cf = frame * CFrame.new(m.c * s) * YAW180
	if not m.cf or (m.cf.Position - cf.Position).Magnitude > 0.01 or m.cf.LookVector:Dot(cf.LookVector) < 0.99995 or m.cf.UpVector:Dot(cf.UpVector) < 0.99995 then
		m.cf = cf; m.p.CFrame = cf
	end
end
local function hide(m) if m and m.tr ~= 1 then m.tr = 1; m.p.Transparency = 1 end end
local function kill(list) for _, m in list do if m and m.p then m.p:Destroy() end end end
-- a spec curve: { { u, value }, ... } eased between keys
local function curve(keys, u)
	if u <= keys[1][1] then return keys[1][2] end
	for i = 1, #keys - 1 do
		local a, b = keys[i], keys[i + 1]
		if u <= b[1] then return a[2] + (b[2] - a[2]) * smooth((u - a[1]) / (b[1] - a[1])) end
	end
	return keys[#keys][2]
end
local function flat(v) local f = Vector3.new(v.X, 0, v.Z); return (f.Magnitude > 1e-3) and f.Unit or Vector3.new(0, 0, -1) end
local function onG(ctx, p) return Vector3.new(p.X, ctx.ground(p.X, p.Z), p.Z) end
-- the cast frame: fixed at t = 0 on the ground under the Titan, facing its cast direction (spec: aims lock at the cast)
local function castAt(ctx)
	local o = ctx.base.Position
	local g = onG(ctx, o)
	ctx.tcast = CFrame.lookAt(g, g + flat(ctx.base.LookVector))
	return ctx.tcast
end
local function cast(ctx) return ctx.tcast or castAt(ctx) end
local function groundFrame(ctx, p, look) local g = onG(ctx, p); return CFrame.lookAt(g, g + flat(look)) end

-- K3 debris: deterministic cosmetic paths from the real contact points (spec 8.5), K4 drops ride the same way
local DEBRIS = {
	slam = { n = 7, size = { 0.35, 1.2 }, out = { 7, 13 }, up = { 9, 16 }, spin = { 180, 400 }, life = 0.85 },
	bomb = { n = 5, size = { 0.30, 0.85 }, out = { 5, 11 }, up = { 7, 14 }, spin = { 150, 350 }, life = 0.75 },
	wave = { n = 9, size = { 0.45, 1.4 }, out = { 8, 16 }, up = { 10, 18 }, spin = { 200, 450 }, life = 1.10 },
}
local function debris(ctx, kind, origins, away: Vector3?, t0, seed)
	local D = DEBRIS[kind]
	local rnd = Random.new(seed)
	local list = {}
	for i = 1, D.n do
		local o = origins[(i - 1) % #origins + 1]
		local m = mesh(ctx, "titanRock" .. ((i - 1) % 4 + 1))
		if not m then break end
		local a = (away and math.atan2(away.Z, away.X) or rnd:NextNumber(0, 2 * pi)) + rnd:NextNumber(-1.1, 1.1)
		local sp = rnd:NextNumber(D.out[1], D.out[2])
		list[i] = { m = m, o = o + Vector3.new(0, 0.3, 0), v = Vector3.new(cos(a) * sp, rnd:NextNumber(D.up[1], D.up[2]), sin(a) * sp),
			s = rnd:NextNumber(D.size[1], D.size[2]), axis = rnd:NextUnitVector(), w = rad(rnd:NextNumber(D.spin[1], D.spin[2])) }
	end
	-- a few drops along with the rocks (K4)
	for i = 1, 3 do
		local m = mesh(ctx, "titanDrop" .. i)
		if not m then break end
		local a = rnd:NextNumber(0, 2 * pi); local o = origins[(i - 1) % #origins + 1]
		table.insert(list, { m = m, drop = true, o = o + Vector3.new(0, 0.5, 0), v = Vector3.new(cos(a) * 5, 12, sin(a) * 5), s = 1, axis = Vector3.yAxis, w = 0 })
	end
	return { step = function(_, t)
		local e = t - t0
		if e > D.life then return false end
		for _, d in list do
			local p = d.o + d.v * e + Vector3.new(0, -14 * e * e, 0)   -- (cosmetic gravity 28)
			local gy = onG(ctx, p).Y
			if p.Y < gy + 0.2 then p = Vector3.new(p.X, gy + 0.2, p.Z) end
			local look = (d.drop and d.v + Vector3.new(0, -28 * e, 0)) or nil
			local cf = look and CFrame.lookAt(p, p + look) * CFrame.Angles(-pi / 2, 0, 0) or (CFrame.new(p) * CFrame.fromAxisAngle(d.axis, d.w * e))
			if not d.sized then d.sized = true; d.m.p.Size = d.m.s0 * d.s; d.m.p.Transparency = 0 end   -- (size once; perf note above)
			d.m.p.CFrame = cf * YAW180
			if e > D.life - 0.15 then d.m.p.Transparency = smooth((e - (D.life - 0.15)) / 0.15) end
		end
		return true
	end, stop = function() for _, d in list do d.m.p:Destroy() end end }
end

-- ===== keys: one spec row { t, ease, Y, F, P, bodyRx, bodyOy, topRx, topOy, lid } + tentacle rows
-- { rx = { s1, s2, s3 }, rz = { s1, s2, s3 } } for the front and the back pairs (spec tables, +X side; the -X side mirrors)
local EZ = { E = "smooth", O = "out", I = "in", L = "lin" }
local function keys(rows, front, back)
	assert(#front == #rows and #back == #rows, "Titan key rows mismatch")
	local out = {}
	for i, r in rows do
		local b = {}
		b.Body = { -r[6], 0, 0, 0, r[7], 0 }
		b.Top = { -r[8], 0, 0, 0, r[9], 0 }
		b.Lid = { LID_FULL * r[10] / 60, 0, 0 }
		local f, k = front[i], back[i]
		for s = 1, 3 do
			b["TentFL" .. s] = { -f[1][s], 0, f[2][s] }
			b["TentFR" .. s] = { -f[1][s], 0, -f[2][s] }
			b["TentBL" .. s] = { 0, k[1][s], -k[2][s] }
			b["TentBR" .. s] = { 0, -k[1][s], k[2][s] }
		end
		out[i] = { t = r[1], ease = EZ[r[2]] or "smooth", root = { r[3], 0, r[4], -r[5], 0, 0 }, bones = b }
	end
	return out
end
local Z3 = { 0, 0, 0 }
local function T(rx, rz) return { rx or Z3, rz or Z3 } end
local function mix3(a, b, u) return { a[1] + (b[1] - a[1]) * u, a[2] + (b[2] - a[2]) * u, a[3] + (b[3] - a[3]) * u } end
local function mixT(a, b, u) return { mix3(a[1], b[1], u), mix3(a[2], b[2], u) } end

-- ===== the world-space tip contact (spec 4.4 / 6.3): after the pose, each chain's end bone (its "3" bone, the contact
-- point the batch 2 solver and James's approval used) is pulled onto a ground target by its joints 2 and 1, at most 16 deg
-- each x the weight (the spec's 8 deg per joint per pass, two passes). An exact two-bone solve: joint 2 bends to the
-- target's distance (law of cosines), then joint 1 aims (measured: a CCD only turns toward the target and could not
-- shorten a reach, 0.6-2.2 studs left over)
local MAXC = rad(16)
local MAXC_BACK = rad(30)    -- (the back chains: the spec's own caveat, "rear tips require a proper reach test"; their rest
                             -- tips curl UP beside the body, 4.5 studs from a spot the 9-stud chain must fold onto)
local function turnBone(e, q: CFrame)
	local W = e.b.TransformedWorldCFrame
	local P = W * e.b.Transform:Inverse()
	e.b.Transform = P:Inverse() * (CFrame.new(W.Position) * q * W.Rotation)
end
local function angleOf(a: Vector3, b: Vector3) return math.acos(math.clamp(a.Unit:Dot(b.Unit), -1, 1)) end
local function reachTip(bones, chain, target: Vector3, w)
	local e1, e2, e3 = bones[chain .. "1"], bones[chain .. "2"], bones[chain .. "3"]
	if not (e1 and e2 and e3) or w <= 0 then return nil end
	local cap = chain:find("B") and MAXC_BACK or MAXC
	local J1, J2, Tp = e1.b.TransformedWorldCFrame.Position, e2.b.TransformedWorldCFrame.Position, e3.b.TransformedWorldCFrame.Position
	local l1, l2 = (J2 - J1).Magnitude, (Tp - J2).Magnitude
	local d = math.clamp((target - J1).Magnitude, math.abs(l1 - l2) + 0.05, l1 + l2 - 0.05)
	local n = (J1 - J2):Cross(Tp - J2)
	if n.Magnitude > 1e-4 then                                  -- (joint 2: open / close the bend to the target's distance)
		local want = math.acos(math.clamp((l1 * l1 + l2 * l2 - d * d) / (2 * l1 * l2), -1, 1))
		local need = want - angleOf(J1 - J2, Tp - J2)
		if e1.b.Parent and e1.b:FindFirstAncestorOfClass("Model"):GetAttribute("FreezeTT") then          -- (tuning readout: the raw pose's need)
			local mdl = e1.b:FindFirstAncestorOfClass("Model")
			local dv = mdl:GetPivot():VectorToObjectSpace(Tp - target)
			mdl:SetAttribute("TNeed" .. chain, string.format("bend %.1f aim %.1f raw %.2f (tip - target, model axes: %.2f, %.2f, %.2f)", math.deg(need),
				math.deg(angleOf(Tp - J1, target - J1)), (Tp - target).Magnitude, dv.X, dv.Y, dv.Z))
		end
		turnBone(e2, CFrame.fromAxisAngle(n.Unit, math.clamp(need, -cap, cap) * w))
	end
	Tp = e3.b.TransformedWorldCFrame.Position                  -- (joint 1: aim the chain at the target)
	local a, b = Tp - J1, target - J1
	local ax = a:Cross(b)
	if ax.Magnitude > 1e-4 then turnBone(e1, CFrame.fromAxisAngle(ax.Unit, math.min(angleOf(a, b), cap) * w)) end
	return (e3.b.TransformedWorldCFrame.Position - target).Magnitude
end
local function window(t, t0, t1, t2, t3)
	if t <= t0 or t >= t3 then return 0 end
	if t < t1 then return smooth((t - t0) / (t1 - t0)) end
	if t <= t2 then return 1 end
	return 1 - smooth((t - t2) / (t3 - t2))
end
local function contacts(list, t0, t1, t2, t3)
	return function(ctx, t, bones)
		local w = window(t, t0, t1, t2, t3)
		if w <= 0 or not ctx.tcast then return end
		local worst = 0
		for _, c in list do
			local p = ctx.tcast * c[2]
			local err = reachTip(bones, c[1], Vector3.new(p.X, ctx.ground(p.X, p.Z) + (c[3] or 0.12), p.Z), w)
			if err then worst = math.max(worst, err) end
		end
		if ctx.m:GetAttribute("FreezeTT") then ctx.m:SetAttribute("TContactErr", worst) end   -- (test readout)
	end
end

-- the mouth (batch 2 measurement: 6.7 ahead of the Top bone and 2.07 up at rest), carried by the Top bone's real turn
-- (r2, ChatGPT: a fixed offset left the ball off the mouth while the head leaned / aimed)
local function restCF(b: Bone): CFrame
	local cf = b.CFrame
	local p = b.Parent
	while p and p:IsA("Bone") do cf = p.CFrame * cf; p = p.Parent end
	return cf
end
local MOUTH = Vector3.new(0, 2.07, -6.7)
local function mouthAt(ctx)
	local top = ctx.topBone or ctx.m:FindFirstChild("Top", true)
	ctx.topBone = top
	local mesh = top and top:FindFirstAncestorWhichIsA("BasePart")
	if not mesh then return ctx.bonePos("Top") + Vector3.new(0, 2.07, 0) end
	local W = top.TransformedWorldCFrame
	local restW = mesh.CFrame * restCF(top)
	return W.Position + (W.Rotation * restW.Rotation:Inverse()) * ctx.m:GetPivot():VectorToWorldSpace(MOUTH)   -- (MOUTH is in model axes)
end

	for i = 1, #BOMB_PATH - 1 do if t <= BOMB_PATH[i + 1][1] then k = i; break end end

	local ha = a[3] or (from.Y - gyFrom)
	-- (height: eased up to the apex at 3.00 with a flat tangent there, then a steep fall)
	local hu = (k <= 2) and (1 - (1 - u) ^ 2) or (u * u)
	if k == 1 then hu = 1 - (1 - u) ^ 1.5 end
	local h = ha + (b[3] - ha) * hu
	local xz = from:Lerp(hit, f)
	return Vector3.new(xz.X, 0, xz.Z), h
end

local function hurtKeys()
	local out = {}
	for i, r in HURT do
		local b = { Body = { 0, 0, r[6], r[7], 0, 0 }, Top = { 0, 0, r[8], r[9], 0, 0 }, Lid = { LID_FULL * r[10] / 60, 0, 0 } }

			b["TentFL" .. s] = { -OPP_RX[i][s], 0, -HIT_RZ[i][s] }     -- (the far side: the mirrored sign, flung up / out)
			b["TentBL" .. s] = { 0, BACK_RX[i][s], 0 }

		end
		out[i] = { t = r[1], ease = EZ[r[2]], root = { r[3], r[4], 0, 0, r[5], 0 }, bones = b }
	end
	return out
end
